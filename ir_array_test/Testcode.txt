/*
 * =============================================================================
 *  LINE FOLLOWER — v10 (production)
 * =============================================================================
 *  Target   : ESP32 DevKit V1, L298N dual H-bridge, 5-channel TCRT5000L array
 *  Platform : PlatformIO / Arduino framework
 *
 *      [env:esp32doit-devkit-v1]
 *      platform = espressif32
 *      board = esp32doit-devkit-v1
 *      framework = arduino
 *      monitor_speed = 115200
 *
 *  Console: 'g' go, 'i' stop, 'f' speed test, 'r' report, 's' stream, 'h' help
 *
 * -----------------------------------------------------------------------------
 *  ARCHITECTURE
 * -----------------------------------------------------------------------------
 *  Six layers, each depending only on the ones above it. No layer reaches
 *  around another, and every cross-layer assumption is resolved in exactly one
 *  place so a sign error cannot hide in two.
 *
 *    1. CONVERSION   volts <-> PWM duty. The single source of truth for speed.
 *    2. MOTORS       physical-side naming, direction, brake vs coast.
 *    3. SENSORS      raw ADC -> validated line position.
 *    4. CONTROL      position -> error -> wheel speeds.
 *    5. RECOVERY     what to do when the line is gone.
 *    6. TELEMETRY    observation only. Compiles out entirely.
 *
 * -----------------------------------------------------------------------------
 *  THE THREE IDEAS THAT MAKE THIS WORK
 * -----------------------------------------------------------------------------
 *  (a) SPEEDS ARE SPECIFIED IN VOLTS, NOT DUTY.
 *      A DC motor responds to voltage. Duty only maps to voltage given a known
 *      supply, and a 4S Li-ion pack swings 16.8 V -> 12.0 V over a charge
 *      cycle. Hardcoding duty therefore means the robot behaves differently
 *      depending on how full the battery is: at one point in development a
 *      fixed cruise duty delivered 4.96 V against a measured 5.10 V start
 *      threshold, and the robot simply would not move. Everything here derives
 *      from V_BATT_NOMINAL through dutyFor(), so one constant fixes all speeds.
 *
 *  (b) TURN RATE IS SET BY THE DIFFERENCE BETWEEN WHEEL SPEEDS.
 *      Not by their absolute values. So steering holds the difference fixed and
 *      slides the pair upward when the inner wheel would fall below the stall
 *      floor. Turn authority is then independent of cruise speed, which is what
 *      makes it safe to slow the robot down for stability. The differences are
 *      themselves in volts, so cornering does not fade as the pack drains.
 *
 *  (c) A WHEEL IS EITHER ABOVE THE STALL FLOOR OR DELIBERATELY BRAKED.
 *      Never in between. A duty below the start threshold is a stalled motor
 *      pretending to be a slow one — it produces no rotation but reads as a
 *      valid command. floorDuty() is the measured threshold plus margin, and
 *      steering clamps to it or brakes outright.
 *
 * -----------------------------------------------------------------------------
 *  MEASURED CONSTANTS — provenance
 * -----------------------------------------------------------------------------
 *  Motor mapping   L298 channel A (EN 13) drives the PHYSICAL RIGHT wheel,
 *                  channel B (EN 33) the LEFT. Established by driving one
 *                  channel and observing which wheel turned. Channel B needs
 *                  its direction inverted; A does not. Both verified forward.
 *  Start threshold 5.10 V, from a duty ramp until first rotation.
 *  Sensor order    S1..S5 = physical left..right, verified.
 *  Sensor output   Digital comparator: ~4095 or ~600, roughly 3400 counts of
 *                  separation, so a fixed 2000 threshold is ample.
 *  Geometry        Array span ~105 mm, pitch ~26 mm, tape 70 mm, axle-to-array
 *                  L = 90 mm.
 *  Motor trim      None needed. Start duties differ (75 vs 85) but with equal
 *                  duty the robot tracks straight, because standstill stiction
 *                  and running speed are governed by different physics.
 *
 * -----------------------------------------------------------------------------
 *  KNOWN LIMITATION
 * -----------------------------------------------------------------------------
 *  Sensor ride height is 25 mm against a datasheet maximum of 12 mm (optimal
 *  3 mm). Signal falls off as roughly 1/d^2, and the IR cone widens with
 *  height, so each sensor averages an ~18 mm patch rather than ~2 mm. Position
 *  is therefore known only to about +/-13 mm regardless of tuning. If cornering
 *  ever needs to be tighter than it is, lowering the array is the fix — no
 *  constant in this file can buy that back.
 * =============================================================================
 */

#include <Arduino.h>
#include <math.h>

/* =============================================================================
 *  BUILD OPTIONS
 * ========================================================================== */

/* Set to 0 for a silent production build. This removes all streaming, the
 * report, and the CSV ring buffer (about 7 kB of RAM). Console commands remain
 * available either way, because 'i' is the emergency stop and must always
 * work. */
#define ENABLE_TELEMETRY 1

/* Begin following automatically after a delay instead of waiting for 'g'.
 * Off by default: a robot that drives the instant it is powered on is a robot
 * that drives off the bench while you are still holding it. */
#define AUTOSTART 0
static const unsigned long AUTOSTART_DELAY_MS = 3000;

/* =============================================================================
 *  CONFIGURATION
 * ========================================================================== */

/* --- Battery. MEASURE AT THE L298N TERMINALS WITH THE MOTORS RUNNING. ------
 * Every speed derives from this. A stale value here does not look like a
 * stale value: it looks like a control bug. Update it whenever you charge. */
static const float V_BATT_NOMINAL = 12.50f;

/* --- Supply chain losses and motor limits --------------------------------- */
static const float V_DROP_L298 = 1.50f;   /* L298 datasheet Table 4, ~350 mA  */
static const float V_MOTOR_MAX = 12.00f;  /* motor rating; never exceed       */

/* --- Speeds, in volts at the motor terminals ------------------------------ */
static const float V_START       = 5.10f; /* MEASURED first-rotation voltage  */
static const float START_MARGIN  = 1.15f; /* floor = V_START * this           */
static const float V_CRUISE      = 6.00f; /* PRIMARY TUNING KNOB             */
static const float V_SPIN        = 6.60f; /* in-place rotation                */

/* --- Steering differentials, in volts ------------------------------------- *
 * The gap between outer and inner wheel at each error level. Raise if the
 * robot drifts before correcting; lower if it weaves. */
static const float DIFF_NEAR_V = 1.30f;   /* |error| == 500                   */
static const float DIFF_FAR_V  = 2.40f;   /* |error| == 1000                  */
                                          /* 1500 -> brake inner wheel        */
                                          /* 2000 -> counter-rotate           */

/* --- Measured chassis speed at V_CRUISE, mm/s ----------------------------- *
 * Press 'f', mark the floor at one wheel before and after, measure between the
 * marks, divide by two. Sets the corner advance duration. Zero falls back to
 * ADVANCE_FALLBACK_MS. */
static const float         ROBOT_SPEED_MMPS = 0.0f;
static const unsigned long ADVANCE_FALLBACK_MS = 300;

/* --- Geometry, mm --------------------------------------------------------- */
static const float SENSOR_AHEAD_MM = 90.0f;  /* axle centre to sensor line    */
static const float LINE_WIDTH_MM   = 70.0f;
static const float RIDE_HEIGHT_MM  = 25.0f;  /* out of spec; see header       */
static const float WHEEL_DIA_MM    = 65.0f;

/* --- Motor pins, named by PHYSICAL side ---------------------------------- */
static const int LEFT_EN = 33, LEFT_IN_A = 27, LEFT_IN_B = 14, LEFT_CH = 1;
static const int RIGHT_EN = 13, RIGHT_IN_A = 25, RIGHT_IN_B = 26, RIGHT_CH = 0;

/* true means positive speed drives that wheel backwards, so negate it. */
static const bool LEFT_INVERT  = true;
static const bool RIGHT_INVERT = false;

static const int PWM_FREQ_HZ = 1000;
static const int PWM_BITS    = 8;

/* --- Sensor pins, physical left to right --------------------------------- *
 * GPIO 36/39/34/35 are input-only with no internal pull resistors: a
 * disconnected wire floats high and reads as "no line". A pattern that never
 * changes usually means a loose wire, not a control problem. */
static const int SENSOR_COUNT = 5;
static const int SENSOR_PINS[SENSOR_COUNT] = { 36, 39, 34, 35, 32 };
static const int SENSOR_THRESHOLD = 2000;
static const bool LINE_READS_LOW  = true;

/* --- Control tuning ------------------------------------------------------ *
 * Position is quantised to multiples of 500 because the sensors are binary, so
 * error takes exactly five magnitudes: 0, 500, 1000, 1500, 2000. DEADBAND must
 * stay below 500 or the entire first correction step is swallowed. */
static const int ERROR_DEADBAND   = 250;
static const int ADVANCE_MAX_ERR  = 1000; /* above this, skip the advance     */

static const unsigned long PIVOT_TIMEOUT_MS = 1500;

/* --- Loop pacing --------------------------------------------------------- *
 * The sampling limit is omega_max = spot / (samples * period * radius). At a
 * ~5 mm optical spot, 3 samples and a 90 mm radius: 4 ms gives 4.6 rad/s, which
 * an in-place spin exceeds; 2 ms gives 9.3 rad/s, which clears every usable
 * speed. When rotation outruns the sensors, sample faster rather than turning
 * slower. */
static const int LOOP_PERIOD_MS = 1;

/* =============================================================================
 *  LAYER 1 — CONVERSION
 *  The single source of truth translating desired volts into PWM duty.
 * ========================================================================== */

/* Volts actually reachable at the motor terminals, after bridge losses. */
static float availableVolts() {
  return V_BATT_NOMINAL - V_DROP_L298;
}

/* Duty required to deliver a target voltage, clamped to the motor rating. */
static int dutyFor(float volts) {
  const float avail = availableVolts();
  if (avail <= 0.1f) return 0;
  if (volts > V_MOTOR_MAX) volts = V_MOTOR_MAX;
  return constrain((int)(255.0f * volts / avail + 0.5f), 0, 255);
}

/* Inverse of dutyFor(), for reporting. */
static float voltsFor(int duty) {
  return availableVolts() * (float)duty / 255.0f;
}

static int ceilingDuty() { return dutyFor(V_MOTOR_MAX); }

/* Lowest duty at which a wheel reliably turns: measured threshold plus margin.
 * Anything below this is a stall, not a slow wheel. */
static int floorDuty() {
  return constrain((int)(dutyFor(V_START) * START_MARGIN), 0, ceilingDuty());
}

static int cruiseDuty() { return constrain(dutyFor(V_CRUISE), 0, ceilingDuty()); }
static int spinDuty()   { return constrain(dutyFor(V_SPIN), floorDuty(), ceilingDuty()); }

/* Distance the chassis must roll after losing the line so that the wheel axle
 * arrives at the corner: the array sits SENSOR_AHEAD_MM in front of the axle,
 * and the outgoing leg's centreline is half a tape width back from the end of
 * the incoming one. */
static float advanceDistanceMm() {
  return SENSOR_AHEAD_MM - LINE_WIDTH_MM / 2.0f;
}

/* That distance expressed as a duration at cruise speed. */
static unsigned long advanceDurationMs() {
  if (ROBOT_SPEED_MMPS > 10.0f)
    return (unsigned long)(advanceDistanceMm() / ROBOT_SPEED_MMPS * 1000.0f);
  return ADVANCE_FALLBACK_MS;
}

/* True when the pack is too low to drive at all. */
static bool supplyTooLow() { return cruiseDuty() <= floorDuty(); }

/* =============================================================================
 *  LAYER 2 — MOTORS
 *  Owns pin polarity and the brake/coast distinction. The only place a
 *  direction sign or a physical side is decided.
 * ========================================================================== */

/* Sentinel speed meaning "hold this wheel still against rotation".
 * L298 datasheet Table 5: both inputs HIGH is a real brake ("fast motor
 * stop"); both LOW merely lets the wheel free-run, and the chassis then drags
 * it. A braked inner wheel gives a tighter and far more repeatable turn. */
static const int WHEEL_BRAKE = -30000;

/* Live wheel commands, exposed for telemetry. */
static int  g_dutyLeft = 0, g_dutyRight = 0;
static bool g_brakeLeft = false, g_brakeRight = false;

static void motorsBegin() {
  pinMode(LEFT_IN_A, OUTPUT);  pinMode(LEFT_IN_B, OUTPUT);
  pinMode(RIGHT_IN_A, OUTPUT); pinMode(RIGHT_IN_B, OUTPUT);
#if ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcAttach(LEFT_EN,  PWM_FREQ_HZ, PWM_BITS);
  ledcAttach(RIGHT_EN, PWM_FREQ_HZ, PWM_BITS);
#else
  ledcSetup(LEFT_CH,  PWM_FREQ_HZ, PWM_BITS); ledcAttachPin(LEFT_EN,  LEFT_CH);
  ledcSetup(RIGHT_CH, PWM_FREQ_HZ, PWM_BITS); ledcAttachPin(RIGHT_EN, RIGHT_CH);
#endif
}

/* Core 2.x addresses LEDC by channel, core 3.x by pin. */
static inline void writePwm(int channel, int duty) {
#if ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcWrite((channel == LEFT_CH) ? LEFT_EN : RIGHT_EN, duty);
#else
  ledcWrite(channel, duty);
#endif
}

/* Drive one bridge. Handles the brake sentinel, inversion and clamping. */
static void driveWheel(int inA, int inB, int channel, int speed, bool invert) {
  if (speed == WHEEL_BRAKE) {
    digitalWrite(inA, HIGH); digitalWrite(inB, HIGH); writePwm(channel, 255);
    return;
  }
  if (invert) speed = -speed;
  const int cap = ceilingDuty();
  speed = constrain(speed, -cap, cap);

  if (speed > 0)      { digitalWrite(inA, HIGH); digitalWrite(inB, LOW);  writePwm(channel,  speed); }
  else if (speed < 0) { digitalWrite(inA, LOW);  digitalWrite(inB, HIGH); writePwm(channel, -speed); }
  else                { digitalWrite(inA, LOW);  digitalWrite(inB, LOW);  writePwm(channel, 0); }
}

/* Command both wheels. Arguments are PHYSICAL left and right, always. */
static void setWheels(int left, int right) {
  g_brakeLeft  = (left  == WHEEL_BRAKE);
  g_brakeRight = (right == WHEEL_BRAKE);
  g_dutyLeft   = g_brakeLeft  ? 0 : left;
  g_dutyRight  = g_brakeRight ? 0 : right;
  driveWheel(LEFT_IN_A,  LEFT_IN_B,  LEFT_CH,  left,  LEFT_INVERT);
  driveWheel(RIGHT_IN_A, RIGHT_IN_B, RIGHT_CH, right, RIGHT_INVERT);
}

static void coastWheels() { setWheels(0, 0); }
static void brakeWheels() { setWheels(WHEEL_BRAKE, WHEEL_BRAKE); }

/* Rotate about the chassis centre. direction > 0 turns right. */
static void spinInPlace(int direction) {
  const int s = spinDuty();
  if (direction > 0) setWheels(s, -s);
  else               setWheels(-s, s);
}

/* =============================================================================
 *  LAYER 3 — SENSORS
 *  Turns five binary channels into one validated line position.
 * ========================================================================== */

/* Weighted position runs 0..4000 across the array; 2000 is centred. */
static const int POSITION_CENTRE = (SENSOR_COUNT - 1) * 1000 / 2;

struct LineReading {
  int  raw[SENSOR_COUNT];
  bool onLine[SENSOR_COUNT];
  bool found;        /* at least one channel sees the line          */
  int  position;     /* 0..4000, valid only when found              */
  int  blackCount;   /* channels reading black, including outliers  */
  int  runStart;     /* first index of the run used                 */
  int  runLength;    /* length of the run used                      */
  bool hadGap;       /* black channels existed outside that run     */
};

static LineReading g_line;
static unsigned long g_gapFrames = 0;

static void sensorsBegin() {
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
}

/* Locate the longest unbroken run of black channels.
 *
 * A real line produces one contiguous run, so this rejects outliers without
 * discarding the frame. Two alternatives were tried and both failed: averaging
 * every black channel lets a single spurious reading at the far end drag the
 * weighted mean badly, while rejecting gapped frames outright means that one
 * systematically weak channel stops the robot entirely. Ties break toward the
 * array centre as the likelier true line. */
static void findLongestRun(LineReading &r) {
  r.runStart = -1;
  r.runLength = 0;
  const float centre = (SENSOR_COUNT - 1) / 2.0f;

  int i = 0;
  while (i < SENSOR_COUNT) {
    if (!r.onLine[i]) { i++; continue; }
    int j = i;
    while (j < SENSOR_COUNT && r.onLine[j]) j++;
    const int length = j - i;

    if (length > r.runLength) {
      r.runLength = length;
      r.runStart  = i;
    } else if (length == r.runLength && r.runStart >= 0) {
      const float midNew = i + (length - 1) / 2.0f;
      const float midOld = r.runStart + (r.runLength - 1) / 2.0f;
      if (fabsf(midNew - centre) < fabsf(midOld - centre)) r.runStart = i;
    }
    i = j;
  }
}

/* Sample every channel and derive the line position. */
static void sampleSensors() {
  LineReading &r = g_line;
  r.blackCount = 0;

  for (int i = 0; i < SENSOR_COUNT; i++) {
    const int value = analogRead(SENSOR_PINS[i]);
    r.raw[i] = value;
    r.onLine[i] = LINE_READS_LOW ? (value < SENSOR_THRESHOLD)
                                 : (value > SENSOR_THRESHOLD);
    if (r.onLine[i]) r.blackCount++;
  }

  findLongestRun(r);

  r.hadGap = (r.runLength > 0) && (r.blackCount != r.runLength);
  if (r.hadGap) g_gapFrames++;

  if (r.runLength == 0) {
    r.found = false;
    r.position = 0;
    return;
  }

  long weighted = 0;
  for (int k = r.runStart; k < r.runStart + r.runLength; k++)
    weighted += (long)k * 1000;

  r.position = (int)(weighted / r.runLength);
  r.found = true;
}

/* True when every channel reads black.
 *
 * A 70 mm tape cannot cover a 105 mm array, so this means either a junction
 * blob genuinely wider than the tape, or the array lifted out of sensing range
 * — out-of-range reads identically to black. Ambiguous, so the caller creeps
 * straight rather than committing to a manoeuvre. */
static bool allChannelsBlack() { return g_line.blackCount == SENSOR_COUNT; }

/* True when the centre channel has the line. This is the only acceptable
 * confirmation that a pivot has finished: an outer channel catching the edge
 * mid-sweep means the line is passing by, not that the robot is centred. */
static bool centreOnLine() { return g_line.onLine[SENSOR_COUNT / 2]; }

/* =============================================================================
 *  LAYER 4 — CONTROL
 *  Position becomes an error, and error becomes wheel speeds.
 * ========================================================================== */

/* Signed lateral error. Positive means the line lies to the robot's RIGHT, so
 * the robot must turn right, which makes the LEFT wheel the outer one. */
static int g_error = 0;

/* Which side the line was last seen on: -1 left, +1 right. Seeds the recovery
 * sweep direction, so it must never be allowed to go stale — it is refreshed on
 * every frame that carries directional information. */
static int g_lastSeenSide = 0;

/* Named for telemetry and for reasoning about traces. */
static const char *g_state = "idle";

/* Convert error magnitude into an outer/inner wheel pair.
 *
 * Idea (b) from the header: hold the DIFFERENCE fixed, because that is what
 * sets turn rate, and slide the pair upward if the inner wheel would drop
 * below the stall floor. Turn authority then does not depend on cruise speed. */
static void applyDifferential(float differentialVolts, bool turnRight,
                              const char *nameRight, const char *nameLeft) {
  const int base  = cruiseDuty();
  const int floorD = floorDuty();
  const int cap   = ceilingDuty();

  const int difference = dutyFor(differentialVolts);
  int outer = base + difference / 2;
  int inner = base - difference / 2;

  if (inner < floorD) { const int lift = floorD - inner; inner += lift; outer += lift; }
  if (outer > cap)    { const int drop = outer - cap;    outer -= drop; inner -= drop; }
  if (inner < floorD) inner = floorD;   /* range too narrow to hold both */

  g_state = turnRight ? nameRight : nameLeft;
  if (turnRight) setWheels(outer, inner);
  else           setWheels(inner, outer);
}

/* Map the five possible error magnitudes onto five responses.
 *
 * Five explicit levels rather than a continuous gain, because binary sensors
 * quantise position to multiples of 500 and a smooth Kp buys nothing at that
 * resolution. What levels do buy is idea (c): each one guarantees every wheel
 * is above the floor or deliberately braked, never stalled by accident. */
static void applySteering(int error) {
  const int magnitude = abs(error);
  const bool turnRight = (error > 0);

  if (magnitude <= ERROR_DEADBAND) {
    const int base = cruiseDuty();
    setWheels(base, base);
    g_state = "straight";
    return;
  }

  if (magnitude > 1500) {              /* line under an outer channel only */
    spinInPlace(turnRight ? 1 : -1);
    g_state = turnRight ? "spin R" : "spin L";
    return;
  }

  if (magnitude > 1000) {              /* brake the inner wheel outright */
    const int outer = min(cruiseDuty() + dutyFor(DIFF_FAR_V), ceilingDuty());
    if (turnRight) setWheels(outer, WHEEL_BRAKE);
    else           setWheels(WHEEL_BRAKE, outer);
    g_state = turnRight ? "hard R" : "hard L";
    return;
  }

  if (magnitude > 500)
    applyDifferential(DIFF_FAR_V,  turnRight, "turn R", "turn L");
  else
    applyDifferential(DIFF_NEAR_V, turnRight, "trim R", "trim L");
}

/* One frame of line following, given that the line is visible. */
static void followLine() {
  if (allChannelsBlack()) {
    const int base = cruiseDuty();
    setWheels(base, base);
    g_state = "all-black";
    return;                            /* no directional info; do not update side */
  }

  g_error = g_line.position - POSITION_CENTRE;
  if (g_error > 0)      g_lastSeenSide =  1;
  else if (g_error < 0) g_lastSeenSide = -1;

  applySteering(g_error);
}

/* =============================================================================
 *  LAYER 5 — RECOVERY
 *  A four-phase state machine for a line that has disappeared.
 *
 *    Track   -> following normally
 *    Advance -> roll forward so the axle reaches the corner, then pivot
 *    PivotA  -> rotate toward the last-seen side until the centre confirms
 *    PivotB  -> rotate the other way, given twice as long
 *    Halted  -> terminal; brakes held until re-armed
 * ========================================================================== */

enum Phase { PHASE_TRACK, PHASE_ADVANCE, PHASE_PIVOT_A, PHASE_PIVOT_B, PHASE_HALTED };

static Phase g_phase = PHASE_TRACK;
static unsigned long g_phaseStartMs = 0;
static int g_pivotDirection = 1;

static void enterAdvance() {
  g_phase = PHASE_ADVANCE;
  g_phaseStartMs = millis();
  const int base = cruiseDuty();
  setWheels(base, base);
  g_state = "lost:adv";
}

static void enterPivotA() {
  g_phase = PHASE_PIVOT_A;
  g_phaseStartMs = millis();
  g_pivotDirection = (g_lastSeenSide >= 0) ? 1 : -1;
  spinInPlace(g_pivotDirection);
  g_state = "lost:pivA";
}

static void enterPivotB() {
  g_phase = PHASE_PIVOT_B;
  g_phaseStartMs = millis();
  g_pivotDirection = -g_pivotDirection;
  spinInPlace(g_pivotDirection);
  g_state = "lost:pivB";
}

static void enterHalted() {
  g_phase = PHASE_HALTED;
  brakeWheels();
  g_state = "HALTED";
#if ENABLE_TELEMETRY
  Serial.println(F("HALTED: swept both ways, centre channel never confirmed."));
  Serial.println(F("Press 'i' then 'g' to re-arm."));
#endif
}

static void resumeTracking() {
  g_phase = PHASE_TRACK;
  followLine();
}

/* Advance only makes sense if the line was lost from a roughly centred,
 * straight-ahead state — the array running off the end of the tape at a corner,
 * with the axle still behind the junction. If the robot was already far off the
 * line, rolling forward carries it further away and the subsequent pivot cannot
 * reach back. Gating on the error at the moment of loss is what fixed cornering. */
static void handleTrackPhase() {
  if (g_line.found) { followLine(); return; }
  if (abs(g_error) <= ADVANCE_MAX_ERR) enterAdvance();
  else                                 enterPivotA();
}

/* Any channel reacquiring is enough here: the line reappearing during the
 * advance means it was never really lost, and the current heading still holds. */
static void handleAdvancePhase() {
  if (g_line.found)                                    { resumeTracking(); return; }
  if (millis() - g_phaseStartMs < advanceDurationMs()) {
    const int base = cruiseDuty();
    setWheels(base, base);
    g_state = "lost:adv";
    return;
  }
  enterPivotA();
}

/* Pivots exit only on the centre channel. See centreOnLine(). */
static void handlePivotAPhase() {
  if (centreOnLine())                                { resumeTracking(); return; }
  if (millis() - g_phaseStartMs < PIVOT_TIMEOUT_MS)  {
    spinInPlace(g_pivotDirection);
    g_state = "lost:pivA";
    return;
  }
  enterPivotB();
}

/* Twice the budget, because this sweep must first undo the previous one before
 * it covers any new ground. */
static void handlePivotBPhase() {
  if (centreOnLine())                                    { resumeTracking(); return; }
  if (millis() - g_phaseStartMs < PIVOT_TIMEOUT_MS * 2)  {
    spinInPlace(g_pivotDirection);
    g_state = "lost:pivB";
    return;
  }
  enterHalted();
}

/* Dispatch one frame of the recovery machine. Every path issues a motor
 * command, so the robot can never be left silently uncommanded. */
static void updateFollowing() {
  switch (g_phase) {
    case PHASE_TRACK:    handleTrackPhase();    break;
    case PHASE_ADVANCE:  handleAdvancePhase();  break;
    case PHASE_PIVOT_A:  handlePivotAPhase();   break;
    case PHASE_PIVOT_B:  handlePivotBPhase();   break;
    case PHASE_HALTED:   brakeWheels(); g_state = "HALTED"; break;
  }
}

/* =============================================================================
 *  OPERATING MODES
 * ========================================================================== */

enum Mode { MODE_IDLE, MODE_FOLLOW, MODE_SPEED_TEST };
static Mode g_mode = MODE_IDLE;
static unsigned long g_speedTestStartMs = 0;
static const unsigned long SPEED_TEST_MS = 2000;

static void enterIdle() {
  g_mode = MODE_IDLE;
  g_phase = PHASE_TRACK;
  coastWheels();
  g_state = "idle";
}

static void updateIdle() {
  coastWheels();
  g_state = "idle";
  /* Report the live error even when parked, so sensor response can be checked
   * by hand without arming the motors. */
  g_error = g_line.found ? (g_line.position - POSITION_CENTRE) : 0;
}

/* Drives straight for a fixed interval so chassis speed can be measured with a
 * ruler. Feeds ROBOT_SPEED_MMPS, which sets the corner advance duration. */
static void updateSpeedTest() {
  if (millis() - g_speedTestStartMs < SPEED_TEST_MS) {
    const int base = cruiseDuty();
    setWheels(base, base);
    g_state = "speedtest";
    return;
  }
  enterIdle();
#if ENABLE_TELEMETRY
  Serial.println(F("Speed test done. Measure between the marks, divide by 2."));
#endif
}

/* =============================================================================
 *  LAYER 6 — TELEMETRY
 *  Observation only. Nothing above this line depends on anything below it, so
 *  setting ENABLE_TELEMETRY to 0 removes it completely.
 * ========================================================================== */

static float g_loopMs = 0, g_loopMsPeak = 0;
static unsigned long g_loopCount = 0;

#if ENABLE_TELEMETRY

static const int  LOG_CAPACITY = 300;
static const int  LOG_EVERY_NTH = 8;
static const unsigned long STREAM_IDLE_MS   = 250;
static const unsigned long STREAM_ACTIVE_MS = 1000;

struct LogEntry {
  uint32_t timeMs;
  uint8_t  bits;
  int16_t  position, error, dutyLeft, dutyRight;
  uint16_t periodUs;
  char     state[10];
};

static LogEntry g_log[LOG_CAPACITY];
static int g_logHead = 0, g_logCount = 0;
static bool g_streaming = true;
static unsigned long g_lastStreamMs = 0;
static const char *g_lastStreamedState = "";

static void logAppend(uint16_t periodUs) {
  LogEntry &e = g_log[g_logHead];
  e.timeMs = millis();
  e.bits = 0;
  for (int i = 0; i < SENSOR_COUNT; i++)
    if (g_line.onLine[i]) e.bits |= (1 << i);
  e.position  = g_line.found ? g_line.position : -1;
  e.error     = g_error;
  e.dutyLeft  = g_dutyLeft;
  e.dutyRight = g_dutyRight;
  e.periodUs  = periodUs;
  strncpy(e.state, g_state, sizeof(e.state) - 1);
  e.state[sizeof(e.state) - 1] = '\0';

  g_logHead = (g_logHead + 1) % LOG_CAPACITY;
  if (g_logCount < LOG_CAPACITY) g_logCount++;
}

static void patternString(char *out) {
  for (int i = 0; i < SENSOR_COUNT; i++) out[i] = g_line.onLine[i] ? '#' : '.';
  out[SENSOR_COUNT] = '\0';
}

/* Kept short deliberately. A 90-character line at 115200 baud blocks for
 * 7.8 ms, which was enough to push the loop period from 2 ms to 13 ms and drop
 * the sampling limit below the spin rate. */
static void streamOneLine() {
  char pattern[SENSOR_COUNT + 1];
  patternString(pattern);
  Serial.printf("%6lu [%s]%s e%5d L%4d%s R%4d%s %s\n",
                (unsigned long)millis(), pattern, g_line.hadGap ? "*" : " ",
                g_error,
                g_dutyLeft,  g_brakeLeft  ? "b" : " ",
                g_dutyRight, g_brakeRight ? "b" : " ",
                g_state);
}

static void printReport() {
  char pattern[SENSOR_COUNT + 1];
  patternString(pattern);

  Serial.println();
  Serial.println(F("================ REPORT ================"));
  Serial.printf("battery set to    : %.2f V   (measure under load)\n", V_BATT_NOMINAL);
  Serial.printf("available at motor: %.2f V   (%.4f V per duty count)\n",
                availableVolts(), availableVolts() / 255.0f);
  Serial.println();
  Serial.println(F("speed           volts   duty"));
  Serial.printf ("start (measured) %6.2f  %5d\n", V_START, dutyFor(V_START));
  Serial.printf ("stall floor      %6.2f  %5d\n", V_START * START_MARGIN, floorDuty());
  Serial.printf ("cruise           %6.2f  %5d   <- primary knob\n", V_CRUISE, cruiseDuty());
  Serial.printf ("spin             %6.2f  %5d\n", V_SPIN, spinDuty());
  Serial.printf ("ceiling          %6.2f  %5d\n", V_MOTOR_MAX, ceilingDuty());
  if (supplyTooLow())
    Serial.println(F("*** cruise is at or below the floor: charge the pack, or"
                     " correct V_BATT_NOMINAL. Following will be refused."));
  Serial.println();
  Serial.println(F("error level    volts   outer/inner"));
  {
    const int base = cruiseDuty(), floorD = floorDuty(), cap = ceilingDuty();
    const float levels[2] = { DIFF_NEAR_V, DIFF_FAR_V };
    for (int i = 0; i < 2; i++) {
      const int difference = dutyFor(levels[i]);
      int outer = base + difference / 2, inner = base - difference / 2;
      if (inner < floorD) { const int lift = floorD - inner; inner += lift; outer += lift; }
      if (outer > cap)    { const int drop = outer - cap;    outer -= drop; inner -= drop; }
      if (inner < floorD) inner = floorD;
      Serial.printf("%9d    %6.2f   %3d / %3d  (difference %d)\n",
                    (i + 1) * 500, levels[i], outer, inner, outer - inner);
    }
    Serial.printf("     1500    %6.2f   %3d / brake\n",
                  DIFF_FAR_V, min(base + dutyFor(DIFF_FAR_V), cap));
    Serial.printf("     2000    spin     %3d / -%d\n", spinDuty(), spinDuty());
  }
  Serial.println();
  Serial.printf("advance          : %lu ms for %.0f mm  (%s)\n",
                advanceDurationMs(), advanceDistanceMm(),
                (ROBOT_SPEED_MMPS > 10.0f) ? "from measured speed"
                                           : "FALLBACK - press 'f' and measure");
  Serial.printf("advance gate     : only when |error| <= %d at loss\n", ADVANCE_MAX_ERR);
  Serial.printf("pivot timeout    : %lu ms\n", PIVOT_TIMEOUT_MS);
  Serial.printf("deadband         : %d  (position quantum is 500)\n", ERROR_DEADBAND);
  Serial.println();
  Serial.println(F("left  wheel: L298 ch B, EN 33, invert true"));
  Serial.println(F("right wheel: L298 ch A, EN 13, invert false"));
  Serial.printf("geometry         : L %.0f mm, tape %.0f mm, height %.0f mm%s\n",
                SENSOR_AHEAD_MM, LINE_WIDTH_MM, RIDE_HEIGHT_MM,
                (RIDE_HEIGHT_MM > 12.0f) ? "  <- out of spec" : "");
  Serial.printf("wheel            : %.0f mm dia, %.1f mm circumference\n",
                WHEEL_DIA_MM, 3.14159f * WHEEL_DIA_MM);
  Serial.println();
  Serial.print(F("raw              : "));
  for (int i = 0; i < SENSOR_COUNT; i++) Serial.printf("%5d ", g_line.raw[i]);
  Serial.println();
  Serial.printf("pattern          : [%s]  run %d/%d  black %d%s  error %d\n",
                pattern, g_line.runStart, g_line.runLength,
                g_line.blackCount, g_line.hadGap ? "  gap" : "", g_error);
  Serial.println();
  Serial.printf("loop period      : %.2f ms now, %.2f ms peak\n", g_loopMs, g_loopMsPeak);
  Serial.printf("gapped frames    : %lu of %lu\n", g_gapFrames, g_loopCount);
  Serial.printf("logged samples   : %d\n", g_logCount);
  Serial.println(F("========================================"));
  Serial.println();
}

static void printCsv() {
  Serial.println();
  Serial.println(F("--- CSV BEGIN ---"));
  Serial.println(F("t_ms,period_us,s1,s2,s3,s4,s5,position,error,duty_l,duty_r,state"));
  const int start = (g_logHead - g_logCount + LOG_CAPACITY) % LOG_CAPACITY;
  for (int k = 0; k < g_logCount; k++) {
    const LogEntry &e = g_log[(start + k) % LOG_CAPACITY];
    Serial.printf("%lu,%u", (unsigned long)e.timeMs, e.periodUs);
    for (int i = 0; i < SENSOR_COUNT; i++) Serial.printf(",%u", (e.bits >> i) & 1);
    Serial.printf(",%d,%d,%d,%d,%s\n",
                  e.position, e.error, e.dutyLeft, e.dutyRight, e.state);
  }
  Serial.println(F("--- CSV END ---"));
  Serial.println();
}

static void printHelp() {
  Serial.println();
  Serial.println(F("g  go        i  stop        f  speed test (2 s)"));
  Serial.println(F("r  report    c  dump CSV    s  stream on/off    h  help"));
  Serial.println();
}

/* Print on state change plus a slow heartbeat, so serial never eats the
 * control loop while the robot is moving. */
static void telemetryUpdate(uint16_t periodUs) {
  if (g_loopCount % LOG_EVERY_NTH == 0) logAppend(periodUs);
  if (!g_streaming) return;

  const unsigned long now = millis();
  const unsigned long gap = (g_mode == MODE_FOLLOW) ? STREAM_ACTIVE_MS : STREAM_IDLE_MS;
  if (g_state != g_lastStreamedState || now - g_lastStreamMs >= gap) {
    g_lastStreamMs = now;
    g_lastStreamedState = g_state;
    streamOneLine();
  }
}

#else   /* telemetry disabled: empty stubs, no code emitted */
static void printReport() {}
static void printCsv() {}
static void printHelp() {}
static void telemetryUpdate(uint16_t) {}
static bool g_streaming = false;
#endif

/* =============================================================================
 *  CONSOLE
 *  Always compiled, even in a silent build: 'i' is the emergency stop.
 * ========================================================================== */

static void startFollowing() {
  if (!g_line.found) {
#if ENABLE_TELEMETRY
    Serial.println(F("Refused: no line under the array."));
#endif
    return;
  }
  if (supplyTooLow()) {
#if ENABLE_TELEMETRY
    Serial.println(F("Refused: cruise is at or below the stall floor."));
    Serial.println(F("Charge the pack, or correct V_BATT_NOMINAL."));
#endif
    return;
  }
  g_mode = MODE_FOLLOW;
  g_phase = PHASE_TRACK;
  g_phaseStartMs = millis();
#if ENABLE_TELEMETRY
  Serial.printf("Following. cruise duty %d (%.2f V), advance %lu ms.\n",
                cruiseDuty(), voltsFor(cruiseDuty()), advanceDurationMs());
#endif
}

static void handleCommand(char c) {
  switch (c) {
    case 'g': startFollowing(); break;
    case 'i': enterIdle();
#if ENABLE_TELEMETRY
              Serial.println(F("Idle."));
#endif
              break;
    case 'f': g_mode = MODE_SPEED_TEST;
              g_speedTestStartMs = millis();
#if ENABLE_TELEMETRY
              Serial.println(F("Driving straight 2 s. Mark the floor at one"
                               " wheel before and after."));
#endif
              break;
    case 'r': printReport(); break;
    case 'c': printCsv();    break;
    case 'h': printHelp();   break;
#if ENABLE_TELEMETRY
    case 's': g_streaming = !g_streaming;
              Serial.printf("Stream %s\n", g_streaming ? "on" : "off");
              break;
#endif
    default: break;
  }
}

static void pollConsole() {
  while (Serial.available()) {
    const char c = (char)Serial.read();
    if (c > ' ') handleCommand(c);      /* ignore CR, LF and spaces */
  }
}

/* =============================================================================
 *  ENTRY POINTS
 * ========================================================================== */

void setup() {
  Serial.begin(115200);
  delay(300);

  sensorsBegin();
  motorsBegin();
  coastWheels();

#if ENABLE_TELEMETRY
  Serial.println();
  Serial.println(F("=== LINE FOLLOWER v10 ==="));
  printReport();
  printHelp();
#endif

#if AUTOSTART
  delay(AUTOSTART_DELAY_MS);
  sampleSensors();
  startFollowing();
#else
  enterIdle();
#endif
}

/* One control cycle: measure elapsed time, take input, sense, act, report,
 * then pace. Sensing happens once per cycle and every consumer reads the same
 * frame, so no two decisions in a cycle can disagree about what was seen. */
void loop() {
  static unsigned long previousUs = micros();
  const unsigned long nowUs = micros();
  const unsigned long periodUs = nowUs - previousUs;
  previousUs = nowUs;

  g_loopMs = periodUs / 1000.0f;
  if (g_loopCount > 200 && g_loopMs > g_loopMsPeak) g_loopMsPeak = g_loopMs;
  g_loopCount++;

  pollConsole();
  sampleSensors();

  switch (g_mode) {
    case MODE_IDLE:       updateIdle();      break;
    case MODE_FOLLOW:     updateFollowing(); break;
    case MODE_SPEED_TEST: updateSpeedTest(); break;
  }

  telemetryUpdate((uint16_t)min(periodUs, 65535UL));

  delay(LOOP_PERIOD_MS);
}
