#include <Arduino.h>

/*
 * LINE FOLLOWER v6 — stop + sweep-search recovery
 * ------------------------------------------------
 * Same steering as v5 (outer sensors = corner spin, inner three = fine steer),
 * but the LOST behavior is rebuilt:
 *   1. Stop briefly (settle + debounce).
 *   2. Sweep-search in place: rotate toward the last-seen side, and if that
 *      misses, reverse and widen the arc (doubling) — an expanding scan that
 *      covers both directions. No forward motion, so it can't drift further off.
 *   3. Line found -> resume normal steering (which re-centers it).
 *   4. Search window exhausted -> stop and stay stopped.
 *
 * With a 7cm line there are no inter-sensor gaps, so all-white always means
 * genuinely off the line -> searching (not coasting) is the right response.
 *
 * Sensors binary ~540 on / 4095 off, threshold 2000. S1..S5 = left..right.
 */

// ================= PIN MAP =================
const int IR_COUNT = 5;
const int LINE_PINS[IR_COUNT] = { 36, 39, 34, 35, 32 };   // S1..S5

const int IN1 = 25, IN2 = 26, IN3 = 27, IN4 = 14;
const int ENA = 13, ENB = 33;
const int CH_A = 0, CH_B = 1;

// ================= PWM =================
const int PWM_FREQ = 1000, PWM_RES = 8;

// ================= SENSOR THRESHOLD =================
const int  LINE_THRESHOLD = 2000;
const bool LINE_IS_LOW    = true;   // on the line, raw value is BELOW threshold

// ================= SPEEDS (duty 0..255) =================
const int SLOW       = 55;    // straight creep
const int MAX_SPEED  = 90;    // cap on driving wheel in a gentle pivot
const int MIN_PWM    = 50;    // stiction floor
const int SPIN_SPEED = 70;    // in-place spin (corners and search)
const float TURN_GAIN = 0.05f;

// Kickstart to break static friction on the first forward move.
const int KICK_PWM = 130;
const int KICK_MS  = 35;

// ================= STEERING =================
const int CENTER_POS = (IR_COUNT - 1) * 1000 / 2;   // = 2000
const int DEADBAND   = 600;

// ================= LOST / SEARCH =================
const int SETTLE_MS        = 120;    // stop-and-settle before searching
const int SEARCH_BASE_MS   = 220;    // first sweep arc duration
const int SEARCH_PHASE_MAX = 1200;   // cap on a single sweep arc
const int SEARCH_MAX_MS    = 2500;   // total search time before giving up

// ================= TIMING =================
const int LOOP_DELAY_MS = 4;

// ================= STATE =================
int  rawValues[IR_COUNT];
bool onLine[IR_COUNT];
int  lastLineSign = 0;         // -1 line last seen left, +1 right
bool creeping = false;         // kickstart edge-detection

// lost/search state
bool wasLost = false;
unsigned long lostStartMs = 0;
bool searching = false;
int  searchDir = 1;            // +1 spin right, -1 spin left
unsigned long phaseStartMs = 0;
long phaseDur = SEARCH_BASE_MS;

// ================= SENSORS =================
void setupSensors() {
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
}

bool readPosition(int &position) {
  long weighted = 0;
  int  count = 0;
  for (int s = 0; s < IR_COUNT; s += 1) {
    rawValues[s] = analogRead(LINE_PINS[s]);
    bool on = LINE_IS_LOW ? (rawValues[s] < LINE_THRESHOLD)
                          : (rawValues[s] > LINE_THRESHOLD);
    onLine[s] = on;
    if (on) { weighted += (long)s * 1000; count += 1; }
  }
  if (count == 0) return false;
  position = (int)(weighted / count);
  return true;
}

// ================= MOTORS =================
int applyFloor(int sp) {
  if (sp > 0 && sp <  MIN_PWM) return  MIN_PWM;
  if (sp < 0 && sp > -MIN_PWM) return -MIN_PWM;
  return sp;
}

void setMotor(int inA, int inB, int ch, int sp) {
  sp = constrain(sp, -255, 255);
  if (sp > 0)      { digitalWrite(inA, HIGH); digitalWrite(inB, LOW);  ledcWrite(ch, sp);  }
  else if (sp < 0) { digitalWrite(inA, LOW);  digitalWrite(inB, HIGH); ledcWrite(ch, -sp); }
  else             { digitalWrite(inA, LOW);  digitalWrite(inB, LOW);  ledcWrite(ch, 0);   }
}

void setWheels(int l, int r) {
  setMotor(IN1, IN2, CH_A, l);
  setMotor(IN4, IN3, CH_B, r);   // IN4/IN3 order corrects right-motor polarity
}

void stopMotors() { setWheels(0, 0); }

void spinInPlace(int dir) {        // dir +1 = right, -1 = left
  if (dir > 0) setWheels(SPIN_SPEED, -SPIN_SPEED);
  else         setWheels(-SPIN_SPEED, SPIN_SPEED);
}

void kickForward(int s) {
  if (creeping) return;
  setMotor(IN1, IN2, CH_A, KICK_PWM);
  setMotor(IN4, IN3, CH_B, KICK_PWM);
  delay(KICK_MS);
  creeping = true;
}

// ================= SETUP =================
void setup() {
  Serial.begin(115200);
  delay(300);
  setupSensors();

  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  ledcSetup(CH_A, PWM_FREQ, PWM_RES); ledcAttachPin(ENA, CH_A);
  ledcSetup(CH_B, PWM_FREQ, PWM_RES); ledcAttachPin(ENB, CH_B);
  stopMotors();

  Serial.println("\n=== LINE FOLLOWER v6 (stop + sweep search) ===");
}

// ================= MAIN LOOP =================
void loop() {
  int position;
  bool seen = readPosition(position);
  bool s1 = onLine[0];
  bool s5 = onLine[4];
  const char *state = "";

  if (seen) {
    wasLost = false;
    searching = false;

    if (s1 && !s5) {
      creeping = false;
      spinInPlace(-1);                          // hard LEFT
      lastLineSign = -1; state = "spin L";
    } else if (s5 && !s1) {
      creeping = false;
      spinInPlace(1);                           // hard RIGHT
      lastLineSign = 1; state = "spin R";
    } else if (s1 && s5) {
      int sp = applyFloor(SLOW);
      kickForward(sp);
      setWheels(sp, sp); state = "junction";    // both outer black -> creep
    } else {
      int error = position - CENTER_POS;
      int mag   = abs(error);
      if (mag <= DEADBAND) {
        int sp = applyFloor(SLOW);
        kickForward(sp);
        setWheels(sp, sp); state = "straight";
      } else {
        creeping = false;
        int drive = applyFloor(constrain(SLOW + (int)(TURN_GAIN * (mag - DEADBAND)),
                                         SLOW, MAX_SPEED));
        if (error > 0) { setWheels(drive, 0); state = "gentle R"; }
        else           { setWheels(0, drive); state = "gentle L"; }
      }
      if (error > 0) lastLineSign = 1;
      else if (error < 0) lastLineSign = -1;
    }
  } else {
    // ---- LOST: stop, then expanding sweep-search ----
    unsigned long now = millis();
    if (!wasLost) {                             // just became lost -> init
      wasLost = true;
      lostStartMs = now;
      searching = false;
      searchDir = (lastLineSign >= 0) ? 1 : -1; // first guess: last-seen side
      phaseDur = SEARCH_BASE_MS;
      creeping = false;
    }
    unsigned long el = now - lostStartMs;

    if (el < (unsigned long)SETTLE_MS) {
      stopMotors(); state = "lost:stop";
    } else if (el < (unsigned long)SEARCH_MAX_MS) {
      if (!searching) { searching = true; phaseStartMs = now; }
      if (now - phaseStartMs >= (unsigned long)phaseDur) {
        searchDir = -searchDir;                 // reverse and widen the sweep
        phaseDur = min(phaseDur * 2, (long)SEARCH_PHASE_MAX);
        phaseStartMs = now;
      }
      spinInPlace(searchDir); state = "lost:search";
    } else {
      stopMotors(); state = "lost:giveup";      // searched everywhere, stop
    }
  }

  // Throttled debug (comment out once tuned).
  static unsigned long tDbg = 0;
  if (millis() - tDbg > 150) {
    tDbg = millis();
    Serial.printf("on: %d %d %d %d %d  %s\n",
      onLine[0], onLine[1], onLine[2], onLine[3], onLine[4], state);
  }

  delay(LOOP_DELAY_MS);
}
