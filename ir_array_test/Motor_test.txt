#include <Arduino.h>

/*
 * L298N DIRECTION-PIN TEST — ESP32 (jumpers ON, no PWM)
 * -----------------------------------------------------
 * Tests ONLY IN1-IN4. ENA/ENB jumpers stay ON (enable hardwired HIGH),
 * so motors run at full speed and we only exercise direction.
 * Watch Serial Monitor (115200) — each step says what SHOULD happen.
 *
 * WIRING CHECKLIST:
 *   - ENA/ENB jumpers ON (leave them)
 *   - L298N GND <-> ESP32 GND (common ground) — REQUIRED
 *   - Motor supply on +12V / GND
 *   - Wheels OFF the ground
 */

// ---- Direction pins only ----
const int IN1 = 25, IN2 = 26;   // Motor A
const int IN3 = 27, IN4 = 14;   // Motor B

void setup() {
  Serial.begin(115200);
  delay(300);

  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);

  // Start stopped (both direction pins LOW = motor braked/off).
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);

  Serial.println("\n=== L298N DIRECTION TEST (jumpers ON) ===");
  Serial.println("Wheels OFF the ground. Starting in 3s...");
  delay(3000);
}

// Stop both motors.
void allStop() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);
}

void loop() {
  // ---------- MOTOR A ----------
  Serial.println("[A] Motor A FORWARD (IN1=HIGH, IN2=LOW)");
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  delay(2000);
  allStop(); delay(1000);

  Serial.println("[A] Motor A REVERSE (IN1=LOW, IN2=HIGH)");
  digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH);
  delay(2000);
  allStop(); delay(1000);

  // ---------- MOTOR B ----------
  Serial.println("[B] Motor B FORWARD (IN3=HIGH, IN4=LOW)");
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
  delay(2000);
  allStop(); delay(1000);

  Serial.println("[B] Motor B REVERSE (IN3=LOW, IN4=HIGH)");
  digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
  delay(2000);
  allStop(); delay(1000);

  Serial.println("--- cycle complete, repeating in 2s ---\n");
  delay(2000);
}
